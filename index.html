class GuillotineBinPack { constructor(width, height, kerf = 4) { this.binWidth = width; this.binHeight = height; this.kerf = kerf; this.usedRectangles = []; this.freeRectangles = [{ x: 0, y: 0, w: width, h: height }]; } insert(width, height) { let bestIndex = -1; let bestRotated = false; let minShortSide = Infinity; // 1. Хамгийн сайн тохирох чөлөөт загийг олох for (let i = 0; i < this.freeRectangles.length; i++) { let free = this.freeRectangles[i]; // Эргүүлээгүй if (free.w >= width && free.h >= height) { let leftover = Math.min(free.w - width, free.h - height); if (leftover < minShortSide) { minShortSide = leftover; bestIndex = i; bestRotated = false; } } // 90 градус эргүүлсэн if (free.w >= height && free.h >= width) { let leftover = Math.min(free.w - height, free.h - width); if (leftover < minShortSide) { minShortSide = leftover; bestIndex = i; bestRotated = true; } } } if (bestIndex === -1) return null; let free = this.freeRectangles[bestIndex]; let w = bestRotated ? height : width; let h = bestRotated ? width : height; let node = { x: free.x, y: free.y, w: w, h: h, rot: bestRotated }; this.freeRectangles.splice(bestIndex, 1); // 2. БҮРЭН ШУЛУУН ДАЙРЧ ЗҮСЭХ (Guillotine Split) let w_eff = w + this.kerf; let h_eff = h + this.kerf; let remW = free.w - w_eff; let remH = free.h - h_eff; // Хавтангийн хэмжээнээс хамаарч туузыг тэнцүү бүтэн шугамаар тусгаарлана if (remW > 0 && remH > 0) { if (free.w >= free.h) { // Босоо бүтэн зүсэлт this.freeRectangles.push({ x: free.x + w_eff, y: free.y, w: remW, h: free.h }); this.freeRectangles.push({ x: free.x, y: free.y + h_eff, w: w, h: remH }); } else { // Хэвтээ бүтэн зүсэлт this.freeRectangles.push({ x: free.x, y: free.y + h_eff, w: free.w, h: remH }); this.freeRectangles.push({ x: free.x + w_eff, y: free.y, w: remW, h: h }); } } else if (remW > 0) { this.freeRectangles.push({ x: free.x + w_eff, y: free.y, w: remW, h: free.h }); } else if (remH > 0) { this.freeRectangles.push({ x: free.x, y: free.y + h_eff, w: free.w, h: remH }); } this.usedRectangles.push(node); return node; } }

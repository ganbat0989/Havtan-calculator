function importExcelFileForCurrentTab(e) { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(evt) { try { const data = new Uint8Array(evt.target.result); const workbook = XLSX.read(data, { type: 'array' }); const worksheet = workbook.Sheets[workbook.SheetNames[0]]; const rows = XLSX.utils.sheet_to_json(worksheet, { header: 1 }); let importedParts = []; rows.forEach(row => { if (row && row.length >= 2) { let w = parseFloat(row[0]); let h = parseFloat(row[1]); let count = parseInt(row[2]) || 1; if (!isNaN(w) && !isNaN(h) && w > 0 && h > 0) { // hasEdge болон eU1, eU2, eO1, eO2-ийг true болгож тохирууллаа importedParts.push({ w: w, h: h, count: count, hasEdge: true, isDoor: false, doorMinusW: false, doorMinusH: false, eU1: true, eU2: true, eO1: true, eO2: true }); } } }); if (importedParts.length > 0) { const current = sheetsData.find(s => s.id === activeTabId); if (current) { current.parts = importedParts; renderParts(current.parts); autoSave(); clearResults(); alert(`✅ Энэ хавтанд Excel-ээс ${importedParts.length} деталь эмжээрийн тохиргоотойгоор амжилттай орууллаа!`); } } } catch (err) { alert("❌ Excel файлыг уншихад алдаа гарлаа!"); } document.getElementById('excelFileInput').value = ""; }; reader.readAsArrayBuffer(file); }
